"""
trailing.py - Fungsi murni perhitungan Trailing Stop bertingkat (R-multiple).
Aturan:
- SL awal = SL(rugi), berjarak 1R dari entry.
- Saat profit berjalan mencapai 1.5R, SL pindah ke +1R (level 1).
- Tiap tambahan 1R berikutnya (2.5R, 3.5R, ...), SL naik jadi +2R, +3R, dst.
- SL TIDAK PERNAH mundur.
"""
import math


def calculate_trailing_stop(side: int, entry: float, r: float, fav_profit_r: float, current_level: int, current_sl: float, first_trigger_r: float = 1.5) -> tuple[int, float]:
    """
    Menghitung level trailing dan harga SL baru.
    side: 1 (buy) atau -1 (sell)
    entry: harga masuk
    r: jarak SL awal (1R)
    fav_profit_r: profit maksimum berjalan dalam R
    current_level: level trailing saat ini
    current_sl: harga SL saat ini
    """
    lock_gap = first_trigger_r - 1.0  # e.g., 1.5 - 1.0 = 0.5
    new_level = current_level

    if fav_profit_r >= first_trigger_r:
        calculated_level = int(math.floor(fav_profit_r - lock_gap + 1e-9))
        if calculated_level > current_level:
            new_level = calculated_level

    # Hitung harga SL baru berdasarkan level
    if new_level == 0:
        target_sl = entry - r if side == 1 else entry + r
    else:
        target_sl = entry + side * new_level * r

    # Pastikan SL tidak pernah mundur
    if side == 1:
        final_sl = max(current_sl, target_sl)
    else:
        final_sl = min(current_sl, target_sl)

    return new_level, final_sl

"""
risk.py - Kelas dan fungsi murni pengunci kerugian (LossLock) dan filter risiko.
"""
from dataclasses import dataclass
from datetime import date
from typing import Optional


@dataclass
class LossLockState:
    consecutive_losses: int = 0
    locked_date: Optional[date] = None
    override_active: bool = False


class LossLock:
    def __init__(self, max_consecutive_loss: int = 3):
        self.max_consecutive_loss = max_consecutive_loss
        self.state = LossLockState()

    def record_trade_result(self, is_loss_sl: bool, trade_date: date) -> bool:
        """
        Mencatat hasil trade. Jika SL rugi berturut-turut >= max, tandai terkunci untuk trade_date.
        Return True jika sistem terkunci.
        """
        if self.is_locked(trade_date):
            return True

        if is_loss_sl:
            self.state.consecutive_losses += 1
            if self.state.consecutive_losses >= self.max_consecutive_loss:
                self.state.locked_date = trade_date
                self.state.override_active = False
                return True
        else:
            self.state.consecutive_losses = 0

        return False

    def is_locked(self, current_date: date) -> bool:
        """
        Cek apakah sistem sedang terkunci pada tanggal tertentu.
        Jika tanggal berganti, kunci ter-reset otomatis.
        """
        if self.state.locked_date is not None:
            if current_date > self.state.locked_date:
                # Pergantian hari: reset kunci
                self.state.locked_date = None
                self.state.consecutive_losses = 0
                self.state.override_active = False
                return False
            elif current_date == self.state.locked_date:
                return not self.state.override_active

        return False

    def apply_override(self) -> None:
        """Mengaktifkan override manual pengguna (reset ke 0 jatah)."""
        self.state.consecutive_losses = 0
        self.state.override_active = True

"""
state.py - Pengelolaan persis state volatil ke data/state.json secara atomik.
"""
import json
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Optional

BASE_DIR = Path(__file__).resolve().parent.parent
DATA_DIR = BASE_DIR / "data"
STATE_PATH = DATA_DIR / "state.json"


@dataclass
class BotState:
    consecutive_losses: int = 0
    locked_date: Optional[str] = None  # YYYY-MM-DD
    override_active: bool = False
    open_position: Optional[dict] = None
    last_update: Optional[str] = None


def save_state(state: BotState, file_path: Path = STATE_PATH) -> None:
    file_path.parent.mkdir(parents=True, exist_ok=True)
    temp_path = file_path.with_suffix(".tmp")
    data = asdict(state)
    with open(temp_path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)
    temp_path.replace(file_path)


def load_state(file_path: Path = STATE_PATH) -> BotState:
    if not file_path.exists():
        return BotState()
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return BotState(**data)
    except Exception:
        # Tahan terhadap file rusak: buat backup dan kembalikan state bersih
        backup_path = file_path.with_suffix(".bak")
        if file_path.exists():
            file_path.replace(backup_path)
        return BotState()

"""
logger.py - Logging terstruktur dengan rotasi file dan pencatatan transaksi CSV.
"""
import csv
import logging
from logging.handlers import RotatingFileHandler
from pathlib import Path

BASE_DIR = Path(__file__).resolve().parent.parent
LOGS_DIR = BASE_DIR / "logs"
LOG_FILE = LOGS_DIR / "bot.log"
TRADES_CSV = LOGS_DIR / "trades.csv"


def setup_logger() -> logging.Logger:
    LOGS_DIR.mkdir(parents=True, exist_ok=True)
    logger = logging.getLogger("bot_trade")
    logger.setLevel(logging.INFO)

    if not logger.handlers:
        handler = RotatingFileHandler(
            LOG_FILE, maxBytes=2 * 1024 * 1024, backupCount=5, encoding="utf-8"
        )
        formatter = logging.Formatter(
            "%(asctime)s [%(levelname)s] %(message)s", datefmt="%Y-%m-%d %H:%M:%S"
        )
        handler.setFormatter(formatter)
        logger.addHandler(handler)

    return logger


def log_trade_csv(trade_data: dict) -> None:
    LOGS_DIR.mkdir(parents=True, exist_ok=True)
    file_exists = TRADES_CSV.exists()
    headers = [
        "entry_time", "exit_time", "side", "entry_price", "exit_price",
        "label", "level", "pnl_r", "pnl_usd", "spread_points"
    ]
    with open(TRADES_CSV, "a", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=headers, extrasaction="ignore")
        if not file_exists:
            writer.writeheader()
        writer.writerow(trade_data)

"""
test_foundation.py - Unit test untuk Tahap 3 (trailing, risk, state, logger).
"""
import tempfile
from datetime import date
from pathlib import Path

from bot.logger import setup_logger
from bot.risk import LossLock
from bot.state import BotState, load_state, save_state
from bot.trailing import calculate_trailing_stop


def test_trailing_stop_no_rollback():
    # Test Buy position trailing
    entry = 2000.0
    r = 10.0
    
    # 1. Profit 1.0R (belum mencapai first_trigger 1.5R) -> SL tetap SL0 (1990)
    level, sl = calculate_trailing_stop(1, entry, r, 1.0, 0, 1990.0)
    assert level == 0
    assert sl == 1990.0

    # 2. Profit 1.5R -> Level 1 -> SL = 2000 + 1*10 = 2010.0
    level, sl = calculate_trailing_stop(1, entry, r, 1.5, 0, 1990.0)
    assert level == 1
    assert sl == 2010.0

    # 3. Profit turun ke 1.2R -> SL TIDAK BOLEH MUNDUR (tetap 2010.0)
    level, sl = calculate_trailing_stop(1, entry, r, 1.2, 1, 2010.0)
    assert level == 1
    assert sl == 2010.0


def test_loss_lock_and_reset():
    lock = LossLock(max_consecutive_loss=3)
    today = date(2026, 11, 11)
    
    assert not lock.record_trade_result(True, today)   # Loss 1
    assert not lock.record_trade_result(True, today)   # Loss 2
    assert lock.record_trade_result(True, today)       # Loss 3 -> Locked!
    assert lock.is_locked(today)

    # Hari berikutnya -> otomatis unlock
    tomorrow = date(2026, 11, 12)
    assert not lock.is_locked(tomorrow)


def test_state_atomic_save_load():
    with tempfile.TemporaryDirectory() as tmpdir:
        state_file = Path(tmpdir) / "state.json"
        state = BotState(consecutive_losses=2, locked_date="2026-11-11")
        save_state(state, state_file)
        
        loaded = load_state(state_file)
        assert loaded.consecutive_losses == 2
        assert loaded.locked_date == "2026-11-11"
